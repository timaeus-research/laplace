\begin{abstract}
If $\eta(0,\cdot)\ge0$ on the closed noncritical box and $\eta(0,v_1)\ne0$ at some point of the closed
box, continuity produces an interior point where $\eta(0,\cdot)>0$ (the closed box is the closure of the
open box), so the leading coefficient is positive and the asymptotic equivalence
$Z\sim CN^{-p}\log^{d_0}N$ holds. This is the sharp form of the positivity hypothesis recommended in
the third Astra consult: nonnegativity on the minimal face, not on the whole box (already
$\eta(u)=u$ in $d=1$ changes the leading exponent). Zero \texttt{sorry}/\texttt{axiom}.
\end{abstract}

\section{The things formalised}
{\small
\begin{verbatim}
theorem exists_interior_pos {d' : ℕ} (b : ℝ) (hb : 0 < b) (e : (Fin d' → ℝ) → ℝ)
    (hec : Continuous e) (v₁ : Fin d' → ℝ) (hv₁ : ∀ i, 0 ≤ v₁ i ∧ v₁ i ≤ b)
    (hpos : 0 < e v₁) :
    ∃ v₀ : Fin d' → ℝ, (∀ i, 0 < v₀ i ∧ v₀ i < b) ∧ 0 < e v₀
theorem blockCoeff_pos_of_nonneg_ne_zero ...
    (henn : ∀ v : Fin d' → ℝ, (∀ i, 0 ≤ v i ∧ v i ≤ b) → 0 ≤ η 0 v)
    (v₁ : Fin d' → ℝ) (hv₁ : ∀ i, 0 ≤ v₁ i ∧ v₁ i ≤ b) (hne : η 0 v₁ ≠ 0) :
    0 < blockCoeff β b p k k' h' ξ η
theorem blockStateIntegral_isEquivalent_of_ne_zero ... :
    (fun N : ℝ => blockStateIntegral β b N k h k' h' ξ η) ~[atTop]
      fun N : ℝ => blockCoeff β b p k k' h' ξ η * (N ^ (-p) * Real.log N ^ d₀)
\end{verbatim}

\section{Proof strategy}
\texttt{closure\_pi\_set} and \texttt{closure\_Ioo} identify the closed box as the closure of the open
box; \texttt{mem\_closure\_iff} applied to the open set $\{e>0\}$ yields an interior point with $e>0$,
and unit~90's positivity theorem finishes.
